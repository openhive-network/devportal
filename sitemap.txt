layout: null
sitemap: false
permalink: /sitemap.txt
languages: ["en"]
---
{%- assign first_language = site.languages | first -%}
{%- assign default_language = site.default_lang | default: first_language -%}
{%- assign sitemap_urls = '' -%}
{%- for page in site.pages -%}
  {%- unless page.sitemap == false or page.url contains '.xml' or page.url contains '.txt' -%}
    {%- assign raw_path = page.url | replace: 'index.html', '' -%}
    {%- assign last_char = raw_path | slice: -1 -%}
    {%- if last_char == '/' or raw_path contains '.html' -%}
      {%- assign url_path = raw_path -%}
    {%- else -%}
      {%- assign url_path = raw_path | append: '.html' -%}
    {%- endif -%}
    {%- assign page_root = page.path | split: '/' | first -%}
    {%- assign localize_page = true -%}
    {%- if site.exclude_from_localizations contains page_root -%}
      {%- assign localize_page = false -%}
    {%- endif -%}
    {%- assign candidate_url = site.url | append: site.baseurl | append: url_path -%}
    {%- if sitemap_urls == '' -%}
      {%- assign sitemap_urls = candidate_url -%}
    {%- else -%}
      {%- assign sitemap_urls = sitemap_urls | append: '|' | append: candidate_url -%}
    {%- endif -%}
    {%- if localize_page -%}
      {%- for language in site.languages -%}
        {%- unless language == default_language -%}
          {%- if url_path == '/' -%}
            {%- assign localized_path = '/' | append: language | append: '/' -%}
          {%- else -%}
            {%- assign localized_path = '/' | append: language | append: url_path -%}
          {%- endif -%}
          {%- assign candidate_url = site.url | append: site.baseurl | append: localized_path -%}
          {%- assign sitemap_urls = sitemap_urls | append: '|' | append: candidate_url -%}
        {%- endunless -%}
      {%- endfor -%}
    {%- endif -%}
  {%- endunless -%}
{%- endfor -%}
{%- for collection in site.collections -%}
  {%- if collection.output -%}
    {%- for doc in collection.docs -%}
      {%- unless doc.sitemap == false -%}
        {%- assign raw_path = doc.url | replace: 'index.html', '' -%}
        {%- assign last_char = raw_path | slice: -1 -%}
        {%- if last_char == '/' or raw_path contains '.html' -%}
          {%- assign url_path = raw_path -%}
        {%- else -%}
          {%- assign url_path = raw_path | append: '.html' -%}
        {%- endif -%}
        {%- assign candidate_url = site.url | append: site.baseurl | append: url_path -%}
        {%- if sitemap_urls == '' -%}
          {%- assign sitemap_urls = candidate_url -%}
        {%- else -%}
          {%- assign sitemap_urls = sitemap_urls | append: '|' | append: candidate_url -%}
        {%- endif -%}
        {%- for language in site.languages -%}
          {%- unless language == default_language -%}
            {%- if url_path == '/' -%}
              {%- assign localized_path = '/' | append: language | append: '/' -%}
            {%- else -%}
              {%- assign localized_path = '/' | append: language | append: url_path -%}
            {%- endif -%}
            {%- assign candidate_url = site.url | append: site.baseurl | append: localized_path -%}
            {%- assign sitemap_urls = sitemap_urls | append: '|' | append: candidate_url -%}
          {%- endunless -%}
        {%- endfor -%}
      {%- endunless -%}
    {%- endfor -%}
  {%- endif -%}
{%- endfor -%}
{{ sitemap_urls | split: '|' | uniq | sort | join: '
' }}
